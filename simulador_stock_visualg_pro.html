{ Simulador de Stock com Alertas }
{ Estruturas: enquanto, se/senao, leia, escreval }

algoritmo "SimuladorStock"
var
  produto     : caractere
  quantidade : inteiro
  minimo     : inteiro
  entrada    : inteiro
  saida      : inteiro
  opcao      : inteiro
  continuar : logico

inicio
  continuar <- verdadeiro
  escreval("Nome do produto: ")
  leia(produto)
  escreval("Quantidade inicial: ")
  leia(quantidade)
  escreval("Stock mínimo: ")
  leia(minimo)

  enquanto (continuar = verdadeiro) faca
    escreval("1-Entrada 2-Saida 3-Ver 0-Sair")
    leia(opcao)

    se (opcao = 1) entao
      leia(entrada)
      quantidade <- quantidade + entrada
    senao
      se (opcao = 2) entao
        leia(saida)
        se (saida <= quantidade) entao
          quantidade <- quantidade - saida
        senao
          escreval("ERRO: stock insuficiente")
        fimse
      senao
        se (opcao = 0) entao
          continuar <- falso
        fimse
      fimse
    fimse

    se (quantidade <= 0) entao
      escreval("CRITICO: stock zerado!")
    senao
      se (quantidade < minimo) entao
        escreval("AVISO: abaixo do mínimo!")
      fimse
    fimse
  fimenquanto
